@ArcCoherent.mem_diamond_on : ∀ {M : Type u_1} [inst : TopologicalSpace M] {ι : Type u_2} {γ : ι → ℝ → M} {U : Opens M}
  {p q z : M},
  ArcCoherent γ →
    z ∈ (arcCausality γ).diamond U p q →
      z = p ∨
        ∃ i s r t, 0 ≤ s ∧ s ≤ r ∧ r ≤ t ∧ t ≤ 1 ∧ γ i s = p ∧ γ i r = z ∧ γ i t = q ∧ ∀ ρ ∈ Set.Icc s t, γ i ρ ∈ U
@ArcCoherent.causallyConvexIn_on : ∀ {M : Type u_1} [inst : TopologicalSpace M] {ι : Type u_2} {γ : ι → ℝ → M}
  {U : Opens M},
  ArcCoherent γ →
    ∀ {V : Set M},
      V ⊆ ↑U →
        (∀ (i : ι) (s r t : ℝ),
            0 ≤ s → s ≤ r → r ≤ t → t ≤ 1 → (∀ ρ ∈ Set.Icc s t, γ i ρ ∈ U) → γ i s ∈ V → γ i t ∈ V → γ i r ∈ V) →
          (arcCausality γ).CausallyConvexIn V U
