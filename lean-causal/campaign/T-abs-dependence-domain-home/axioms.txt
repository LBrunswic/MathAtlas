@Causality.mem_dependenceDomain : ∀ {M : Type u_1} [inst : TopologicalSpace M] {C : Causality M} {W : Opens M}
  {S : Set M} {p : M},
  p ∈ C.dependenceDomain W S ↔
    p ∈ W ∧ ∀ (I : Set ℝ) (c : ℝ → M), C.Inextensible W I c → p ∈ c '' I → (c '' I ∩ S).Nonempty
