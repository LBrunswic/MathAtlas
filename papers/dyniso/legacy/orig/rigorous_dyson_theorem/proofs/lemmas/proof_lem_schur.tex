% Proof of Lemma \ref{lem:schur}
\begin{proof}
By the block matrix inversion formula, for
\[
M = \begin{pmatrix} A & B \\ C & D \end{pmatrix},
\]
we have $M^{-1}_{11} = (A - BD^{-1}C)^{-1}$ provided $D$ is invertible.

Applying this with $A = zI$, $B = -Y$, $C = -Y^\top$, $D = I$:
\begin{align}
\mcL(z)^{-1}_{11}
&= \bigl(zI - (-Y) \cdot I^{-1} \cdot (-Y^\top)\bigr)^{-1} \\
&= (zI - YY^\top)^{-1}.
\end{align}

Taking the normalized trace gives the Stieltjes transform of $S$.
\end{proof}
