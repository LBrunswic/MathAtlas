% Proof of Lemma \ref{lem:block-inversion}
\begin{proof}
Direct verification by block multiplication. For the $(1,2)$ block:
\begin{align}
(MM^{-1})_{12}
&= A \cdot (-A^{-1}B(D - CA^{-1}B)^{-1}) + B \cdot (D - CA^{-1}B)^{-1} \notag\\
&= -B(D - CA^{-1}B)^{-1} + B(D - CA^{-1}B)^{-1} = 0.
\end{align}
The other blocks follow similarly.
\end{proof}
