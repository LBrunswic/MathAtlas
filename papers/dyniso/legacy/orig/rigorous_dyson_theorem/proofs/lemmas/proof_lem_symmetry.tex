% Proof of Lemma \ref{lem:symmetry}
\begin{proof}
The transpose of $\mcL(z)$ is:
\[
\mcL(z)^\top = \begin{pmatrix} (zI)^\top & (-Y^\top)^\top \\ (-Y)^\top & I^\top \end{pmatrix} = \begin{pmatrix} zI & -Y \\ -Y^\top & I \end{pmatrix} = \mcL(z).
\]

Since $\mcL(z)$ is symmetric, its inverse is also symmetric:
\[
\mcR(z)^\top = (\mcL(z)^{-1})^\top = (\mcL(z)^\top)^{-1} = \mcL(z)^{-1} = \mcR(z).
\]

For the $2N \times 2N$ block matrix $\mcR = \begin{pmatrix} R_{11} & R_{12} \\ R_{21} & R_{22} \end{pmatrix}$, symmetry gives $R_{12} = R_{21}^\top$.

Since $R_{ij} = m_{ij} I_N$ (blocks proportional to identity) by the isotropy structure, we have:
\[
m_{12} I_N = R_{12} = R_{21}^\top = (m_{21} I_N)^\top = m_{21} I_N.
\]

Therefore $m_{12} = m_{21}$.
\end{proof}
