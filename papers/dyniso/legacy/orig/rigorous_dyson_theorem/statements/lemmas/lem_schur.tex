\begin{lemma}[Block Schur Complement Formula]\label{lem:schur}
For the block matrix $\mcL(z) = \begin{pmatrix} zI & -Y \\ -Y^\top & I \end{pmatrix}$, we have
\[
\bigl(\mcL(z)^{-1}\bigr)_{11} = (zI - YY^\top)^{-1}.
\]
In particular, $\frac{1}{N}\Tr\bigl((\mcL(z)^{-1})_{11}\bigr) = G_S(z)$, the Stieltjes transform of $S = YY^\top$.
\end{lemma}
