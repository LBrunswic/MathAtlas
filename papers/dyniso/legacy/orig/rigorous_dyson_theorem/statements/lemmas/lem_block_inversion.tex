\begin{lemma}[Full Block Matrix Inversion]\label{lem:block-inversion}
For a $2 \times 2$ block matrix $M = \begin{pmatrix} A & B \\ C & D \end{pmatrix}$ with invertible $A$ and $D - CA^{-1}B$, the inverse has the form:
\[
M^{-1} = \begin{pmatrix}
(A - BD^{-1}C)^{-1} & -A^{-1}B(D - CA^{-1}B)^{-1} \\
-D^{-1}C(A - BD^{-1}C)^{-1} & (D - CA^{-1}B)^{-1}
\end{pmatrix}.
\]
In particular, the off-diagonal blocks are:
\begin{align}
M^{-1}_{12} &= -A^{-1}B(D - CA^{-1}B)^{-1}, \label{eq:block-inv-12}\\
M^{-1}_{21} &= -D^{-1}C(A - BD^{-1}C)^{-1}. \label{eq:block-inv-21}
\end{align}
\end{lemma}
