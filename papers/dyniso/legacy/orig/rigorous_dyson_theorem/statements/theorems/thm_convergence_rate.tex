\begin{theorem}[Optimal Convergence Rate]\label{thm:convergence-rate}
Under Assumptions (A1)--(A4) of Theorem \ref{thm:main}, for all $z \in \C^+$ with $\Im(z) \geq \eta > 0$,
\[
\E\bigl[\norm{\mcG_N(z) - \bTr(\mcM(z))}^2_F\bigr] \leq \frac{C(\eta, \sigma, \norm{A})}{N^2 \eta^2},
\]
where $C(\eta, \sigma, \norm{A})$ is a constant depending only on the indicated parameters.
\end{theorem}
