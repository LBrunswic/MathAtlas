\begin{lemma}[Resolvent Bounds]\label{lem:resolvent-bounds}
For any self-adjoint matrix $H \in \R^{N \times N}$ and $z \in \C^+$ with $\Im(z) \geq \eta > 0$, the resolvent $R(z) = (zI - H)^{-1}$ satisfies:
\begin{enumerate}
\item $\norm{R(z)} \leq 1/\eta$
\item $\norm{R(z)}_F \leq \sqrt{N}/\eta$
\item $\Im(R(z))$ is positive semidefinite (Herglotz property)
\end{enumerate}
\end{lemma}
