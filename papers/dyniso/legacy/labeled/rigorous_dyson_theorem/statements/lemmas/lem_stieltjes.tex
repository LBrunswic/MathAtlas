\begin{lemma}[Stieltjes Transform Properties]\label{lem:stieltjes}
Let $\mu$ be a probability measure on $\R$ and define its Stieltjes transform
\[
G_\mu(z) = \int_\R \frac{d\mu(\lambda)}{z - \lambda}, \quad z \in \C^+.
\]
Then:
\begin{enumerate}
\item $G_\mu : \C^+ \to \C^+$ is analytic (Herglotz function)
\item $\Im(G_\mu(z)) \cdot \Im(z) > 0$ for all $z \in \C^+$
\item $\lim_{y \to \infty} iy G_\mu(iy) = -1$
\item $G_\mu$ determines $\mu$ uniquely via the Stieltjes inversion formula:
\[
\mu([a, b]) = \lim_{\varepsilon \to 0^+} \frac{1}{\pi} \int_a^b \Im(G_\mu(x + i\varepsilon)) \, dx
\]
\end{enumerate}
\end{lemma}
