\begin{lemma}[Isotropy Moment Relations]\label{lem:isotropy-moments}
Let $X$ satisfy the isotropy condition with parameter $\sigma^2$. Then:
\begin{enumerate}
\item $\E[X_{ij}] = 0$ for all $i, j$
\item $\E[X_{ij} X_{kl}] = \frac{\sigma^2}{N} \delta_{il} \delta_{jk}$
\item For any deterministic matrices $B, C \in \R^{N \times N}$,
\[
\E[\Tr(XBX^\top C)] = \frac{\sigma^2}{N} \Tr(B) \Tr(C) + O(1).
\]
\end{enumerate}
\end{lemma}
