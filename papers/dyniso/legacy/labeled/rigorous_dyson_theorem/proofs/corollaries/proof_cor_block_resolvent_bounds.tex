% Proof of Corollary \ref{cor:block-resolvent-bounds}
\begin{proof}
The matrix $\mcL(z)$ is not self-adjoint, but we can bound its smallest singular value. For any unit vector $(u, v)^\top \in \C^{2N}$,
\begin{align}
\norm{\mcL(z)(u, v)^\top}^2
&= \norm{zu - Yv}^2 + \norm{-Y^\top u + v}^2 \\
&\geq |\Im(z)|^2 \norm{u}^2 \geq \eta^2.
\end{align}
Hence $\sigma_{\min}(\mcL(z)) \geq \eta$, giving $\norm{\mcR(z)} \leq 1/\eta$. The Frobenius bound follows similarly.
\end{proof}
