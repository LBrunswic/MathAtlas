% Proof of Lemma \ref{lem:resolvent-identity}
\begin{proof}
Multiply both sides on the left by $(B - C)$:
\begin{align}
(B - C) \bigl(B^{-1} + B^{-1} C (B-C)^{-1}\bigr)
&= I + C(B-C)^{-1} - C B^{-1} - C B^{-1} C (B-C)^{-1} \\
&= I + C(B-C)^{-1} - C B^{-1}(I + C(B-C)^{-1}) \\
&= I + C(B-C)^{-1} - C B^{-1}(B-C)(B-C)^{-1} \\
&= I + C(B-C)^{-1} - C(B-C)^{-1} = I.
\end{align}
Hence the identity holds.
\end{proof}
