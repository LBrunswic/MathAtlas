% Proof of Lemma \ref{lem:resolvent-bounds}
\begin{proof}
\textbf{(1):} By the spectral theorem, $H = U \Lambda U^\top$ with $U$ orthogonal and $\Lambda$ diagonal. Then
\[
R(z) = U(zI - \Lambda)^{-1}U^\top.
\]
For each eigenvalue $\lambda$ of $H$,
\[
\abs{(z - \lambda)^{-1}}^2 = \frac{1}{|x - \lambda|^2 + y^2} \leq \frac{1}{y^2} = \frac{1}{\eta^2}.
\]
Hence $\norm{R(z)} = \max_j |z - \lambda_j|^{-1} \leq 1/\eta$.

\medskip
\noindent\textbf{(2):} We have
\[
\norm{R(z)}_F^2 = \sum_j \frac{1}{|z - \lambda_j|^2} \leq \frac{N}{\eta^2}.
\]

\medskip
\noindent\textbf{(3):} For $z = x + iy$,
\[
\Im(R(z)) = \Im\bigl(U(zI - \Lambda)^{-1}U^\top\bigr)
= U \cdot \mathrm{diag}\Bigl(\frac{y}{|z - \lambda_j|^2}\Bigr) \cdot U^\top \succeq 0,
\]
since all diagonal entries are positive for $y > 0$.
\end{proof}
