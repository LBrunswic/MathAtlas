% Proof of Proposition \ref{prop:layerwise}
\begin{proof}[Proof sketch]
By concentration of measure, the entries of $D^l$ satisfy
\[
(D^l)_{ii} = \phi'(h^l_i) \approx \E_{h \sim \rho^l}[\phi'(h)] + \text{fluctuations of order } O(1/\sqrt{N}).
\]

For products $(D^l W^l) B (D^l W^l)^\top$, we have
\begin{align}
\E[(D^l W^l) B (D^l W^l)^\top]
&= \E[D^l W^l B W^{l,\top} D^l] \\
&\approx D^l \E[W^l B W^{l,\top}] D^l \\
&= D^l \cdot \frac{\sigma_w^2}{N}\Tr(B) I \cdot D^l \\
&= \frac{\sigma_w^2}{N}\Tr(B) (D^l)^2.
\end{align}

By concentration, $(D^l)^2 \approx \E_{h \sim \rho^l}[\phi'(h)^2] I + O(1/\sqrt{N})$, giving the effective isotropy with variance $\sigma_{eff}^{2,l}$.

The full rigorous proof requires controlling fluctuations using martingale techniques and local laws from random matrix theory (beyond the scope of this paper).
\end{proof}
